بەستەر و کلیلی پرۆژەکەت ئامادەکراوە بۆ خوێندنەوەی ڕاستەوخۆی sales_data

LOCAL + SUPABASE READY
کۆدی SQL بۆ کردنەوەی مۆڵەتی sales_data و دروستکردنی خشتەکانی یوسەر و ئۆفەر: 
-- 1. کردنەوەی مۆڵەتی خوێندنەوەی خشتەی sales_data
ALTER TABLE public.sales_data ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow public read sales_data" ON public.sales_data;
CREATE POLICY "Allow public read sales_data" ON public.sales_data FOR SELECT USING (true);
GRANT SELECT ON public.sales_data TO anon, authenticated;

-- 2. خشتەی بەکارهێنەران و IMEI و چێک لیستی دەسەڵاتەکان
CREATE TABLE IF NOT EXISTS public.app_users (
  id SERIAL PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  full_name TEXT NOT NULL,
  password TEXT NOT NULL,
  imei_device_id TEXT NOT NULL DEFAULT '*',
  permissions JSONB NOT NULL DEFAULT '["price_lookup","market_items","announcements"]',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.app_users ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all app_users" ON public.app_users;
CREATE POLICY "Allow all app_users" ON public.app_users FOR ALL USING (true) WITH CHECK (true);

-- 3. خشتەی ڕاگەیاندنی نرخ و عرض خاص
CREATE TABLE IF NOT EXISTS public.price_announcements (
  id BIGINT PRIMARY KEY,
  product_code TEXT NOT NULL,
  product_name TEXT NOT NULL,
  tiers JSONB NOT NULL,
  condition_text TEXT,
  reason_text TEXT,
  created_by TEXT,
  created_at TEXT
);

ALTER TABLE public.price_announcements ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all price_announcements" ON public.price_announcements;
CREATE POLICY "Allow all price_announcements" ON public.price_announcements FOR ALL USING (true) WITH CHECK (true);
